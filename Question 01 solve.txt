; ========================================================
; Question 1: Count Vowels and Consonants from a String
; EMU8086 / DOS 16-bit Assembly Format
; ========================================================

org 100h

.model small
.stack 100h

.data
    prompt      db 'Enter a string: $'
    msg_vowel   db 0Ah, 0Dh, 'Vowels = $'
    msg_con     db 0Ah, 0Dh, 'Consonants = $'
    
    input_str   db 100 dup('$') ; Buffer to store input string
    
    vowel_cnt   dw 0
    con_cnt     dw 0

.code
main proc
    mov ax, @data
    mov ds, ax

    ; Print prompt message
    mov ah, 09h
    lea dx, prompt
    int 21h

    ; Read string from keyboard (DOS Function 0Ah)
    lea si, input_str
    mov ah, 0Ah
    mov [si], 99h   ; Max buffer size
    int 21h

    ; Point SI to the actual text start (skip length byte)
    inc si
    mov cl, [si]    ; CL contains string length
    cmp cl, 0
    je end_program
    inc si          ; Move SI to the first character

check_loop:
    mov al, [si]
    
    ; Check for carriage return (end of string)
    cmp al, 0Dh
    je display_result
    cmp al, '$'
    je display_result

    ; Convert uppercase letters to lowercase for easy checking
    cmp al, 'A'
    jl next_char
    cmp al, 'Z'
    jg check_lower
    add al, 32      ; Convert to lowercase

check_lower:
    ; Check if it is an alphabet letter (a-z)
    cmp al, 'a'
    jl next_char
    cmp al, 'z'
    jg next_char

    ; Check if Vowel ('a', 'e', 'i', 'o', 'u')
    cmp al, 'a'
    je is_vowel
    cmp al, 'e'
    je is_vowel
    cmp al, 'i'
    je is_vowel
    cmp al, 'o'
    je is_vowel
    cmp al, 'u'
    je is_vowel

    ; If it's a letter but not a vowel, it's a consonant
    inc con_cnt
    jmp next_char

is_vowel:
    inc vowel_cnt

next_char:
    inc si
    dec cl
    jnz check_loop

display_result:
    ; Display Vowels Count
    mov ah, 09h
    lea dx, msg_vowel
    int 21h
    mov ax, vowel_cnt
    call print_number

    ; Display Consonants Count
    mov ah, 09h
    lea dx, msg_con
    int 21h
    mov ax, con_cnt
    call print_number

end_program:
    mov ax, 4C00h
    int 21h
main endp

; Subroutine to print a 16-bit decimal number
print_number proc
    push ax
    push bx
    push cx
    push dx

    mov bx, 10
    xor cx, cx

extract_digits:
    xor dx, dx
    div bx
    push dx
    inc cx
    cmp ax, 0
    jne extract_digits

print_loop:
    pop dx
    add dl, 30h
    mov ah, 02h
    int 21h
    loop print_loop

    pop dx
    pop cx
    pop bx
    pop ax
    ret
print_number endp

end main
